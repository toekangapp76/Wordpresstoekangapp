<script>
(function(){
  var root=document.getElementById('saas-landing');
  if(!root || root.dataset.slReady==='1') return;
  root.dataset.slReady='1';
  var toggle=root.querySelector('.sl-menu-toggle');
  var nav=root.querySelector('.sl-nav-links');
  if(toggle && nav){
    toggle.addEventListener('click',function(){
      var open=nav.classList.toggle('sl-open');
      toggle.setAttribute('aria-expanded',String(open));
      toggle.innerHTML=open?'✕':'☰';
    });
    nav.querySelectorAll('a').forEach(function(link){link.addEventListener('click',function(){nav.classList.remove('sl-open');toggle.setAttribute('aria-expanded','false');toggle.innerHTML='☰';});});
  }
  var track=root.querySelector('.sl-carousel-track');
  var cards=track ? Array.from(track.children) : [];
  var index=0;
  function visibleCount(){return window.matchMedia('(max-width:767px)').matches?1:(window.matchMedia('(max-width:980px)').matches?2:3);}
  function move(){
    if(!track || !cards.length) return;
    var max=Math.max(0,cards.length-visibleCount());
    index=Math.min(Math.max(index,0),max);
    var cardWidth=cards[0].getBoundingClientRect().width;
    var gap=parseFloat(getComputedStyle(track).gap)||0;
    track.style.transform='translateX('+(-index*(cardWidth+gap))+'px)';
  }
  var prev=root.querySelector('.sl-carousel-prev'),next=root.querySelector('.sl-carousel-next');
  if(prev) prev.addEventListener('click',function(){index=Math.max(0,index-1);move();});
  if(next) next.addEventListener('click',function(){var max=Math.max(0,cards.length-visibleCount());index=index>=max?0:index+1;move();});
  window.addEventListener('resize',move);
  move();
})();
</script>
